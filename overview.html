import numpy as np import plotly.graph_objects as go # ------------------------ # Example orbit # ------------------------ theta = np.linspace(0, 2*np.pi, 60) # 60 steps sat_x = 1.5 * np.cos(theta) sat_y = 1.5 * np.sin(theta) sat_z = 0.5 * np.sin(2*theta) # ------------------------ # Base Earth Sphere (reduced resolution for speed) # ------------------------ u = np.linspace(0, 2*np.pi, 50) # fewer points v = np.linspace(0, np.pi, 50) x0 = np.outer(np.cos(u), np.sin(v)) y0 = np.outer(np.sin(u), np.sin(v)) z0 = np.outer(np.ones(np.size(u)), np.cos(v)) def rotate_earth(x, y, angle): """Rotate Earth around z-axis""" x_rot = x*np.cos(angle) - y*np.sin(angle) y_rot = x*np.sin(angle) + y*np.cos(angle) return x_rot, y_rot # ------------------------ # Frames (Earth rotation + satellite motion) # ------------------------ frames = [] for k in range(len(theta)): angle = 2*np.pi * (k / len(theta)) * 0.25 # quarter spin per orbit x_rot, y_rot = rotate_earth(x0, y0, angle) frames.append(go.Frame( data=[ go.Surface( x=x_rot, y=y_rot, z=z0, colorscale="Earth", showscale=False, opacity=0.95 ), go.Scatter3d( x=[sat_x[k]], y=[sat_y[k]], z=[sat_z[k]], mode="markers+lines", marker=dict(size=6, color="yellow"), line=dict(color="yellow", width=2), name="Satellite" ) ], name=f"frame{k}" )) # ------------------------ # Initial objects # ------------------------ earth_init = go.Surface( x=x0, y=y0, z=z0, colorscale="Earth", showscale=False, opacity=0.95 ) satellite = go.Scatter3d( x=[sat_x[0]], y=[sat_y[0]], z=[sat_z[0]], mode="markers", marker=dict(size=6, color="yellow"), name="Satellite" ) # ------------------------ # Layout # ------------------------ fig = go.Figure(data=[earth_init, satellite], frames=frames) fig.update_layout( margin=dict(l=0, r=0, t=30, b=0), title="🌍 Lightweight Globe with Satellite Orbit", paper_bgcolor="black", scene=dict( xaxis=dict(visible=False), yaxis=dict(visible=False), zaxis=dict(visible=False), aspectmode="data", bgcolor="black" ), updatemenus=[dict( type="buttons", showactive=False, buttons=[ dict(label="▶ Play", method="animate", args=[None, {"frame": {"duration":100, "redraw":True}, "fromcurrent":True}]), dict(label="⏸ Pause", method="animate", args=[[None], {"frame": {"duration":0, "redraw":False}, "mode":"immediate"}]) ], x=0.1, y=1.15, bgcolor="black", font=dict(color="white") )], ) # ------------------------ # Save smaller HTML # ------------------------ fig.write_html("lightweight_satellite_globe.html", include_plotlyjs="cdn") print("✅ Lightweight globe saved as lightweight_satellite_globe.html")
